# ✅ 1. Test Class (Main Flow)

```java
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;
import org.testng.Assert;
import org.testng.annotations.*;

public class TestEcommerce {

    WebDriver driver;
    LoginPage loginPage;
    HomePage homePage;
    CartPage cartPage;

    @BeforeClass
    public void setup() {
        driver = new ChromeDriver();
        driver.get("https://example.com");

        loginPage = new LoginPage(driver);
        homePage = new HomePage(driver);
        cartPage = new CartPage(driver);
    }

    @Test
    public void test_flow() {

        // 🔹 Login
        loginPage.login("testuser", "password");

        // 🔹 Add to Cart
        homePage.addToCart();

        // 🔹 Assert Item Added
        Assert.assertTrue(cartPage.isItemPresent(), "Item not added to cart");

        // 🔹 Logout
        homePage.logout();
    }

    @AfterClass
    public void tearDown() {
        driver.quit();
    }
}
```

# ✅ 2. Login Page (POM)

```java
import org.openqa.selenium.*;

public class LoginPage {

    WebDriver driver;

    By username = By.id("username");
    By password = By.id("password");
    By loginBtn = By.id("loginBtn");

    public LoginPage(WebDriver driver){
        this.driver = driver;
    }

    public void login(String user, String pass){
        driver.findElement(username).sendKeys(user);
        driver.findElement(password).sendKeys(pass);
        driver.findElement(loginBtn).click();
    }
}
```

# ✅ 3. Home Page (Add to Cart + Logout)

```java
import org.openqa.selenium.*;

public class HomePage {

    WebDriver driver;

    By addToCartBtn = By.id("addToCart");
    By logoutBtn = By.id("logout");

    public HomePage(WebDriver driver){
        this.driver = driver;
    }

    public void addToCart(){
        driver.findElement(addToCartBtn).click();
    }

    public void logout(){
        driver.findElement(logoutBtn).click();
    }
}
```

---

# ✅ 4. Cart Page (Assertion)

```java
import org.openqa.selenium.*;

public class CartPage {

    WebDriver driver;

    By cartItem = By.id("cartItem");

    public CartPage(WebDriver driver){
        this.driver = driver;
    }

    public boolean isItemPresent(){
        return driver.findElement(cartItem).isDisplayed();
    }
}
```
